\chapter{Base change and permanent models}\label{ch:base-change-and-permanent-models}
\chaptermark{Base change and permanent models}

This chapter studies the behaviour of models and their component valuations under finite extensions of the base field. Its main themes are normalized base change, reduced special fibres and permanent models, the elimination of ramification, and illustrative tame and wild examples.

Throughout this chapter, $X$ is a smooth, projective and absolutely irreducible curve over the local field $K$.

\section{Normalized base change}\label{sec:normalized-base-change}
We want to understand the behaviour of models under base change with respect to an extension of the base field.
Usually, $L|K$ will denote a finite field extension.
Since $v_K$ is Henselian, it has a unique extension $v_L$ to $L$, and $(L,v_L)$ also satisfies Assumption \ref{ass:v_K}.
Given a model $\X$ of $X$, we ask whether the base change $\X_{\OL}$ is again a model of $X_L$.
The base change satisfies all the conditions in Definition \ref{def:model} except possibly normality.
We obtain a model of $X_L$, called the {\em normalized base change} of $\X$ relative to $L|K$, by normalizing $\X_{\OL}$.\footnote{
    Here we use the fact that Assumption \ref{ass:v_K} implies that $\OO_K$ is an \emph{excellent} ring.
    It follows that $\X_{\OL}$ is an excellent scheme, and therefore its normalization is a finite morphism.
    See \cite[Chapter 8, in particular Theorem 8.2.39 (d)]{Liu}.
}

\section{Reduced special fibres and permanence}\label{sec:reduced-special-fibres-and-permanence}
We introduce permanent models and examine their relation to reduced special fibres and weakly unramified component valuations.


\begin{definition} \label{def:permanent-model}
    A model $\X$ of $X$ is called \textit{permanent} if $\X_{\OL}$ is normal for all finite field extensions $L|K$.
\end{definition}

The main permanence statement is as follows.

\begin{theorem} \label{thm:permanence}
    \begin{enumerate}[label=\textup{(\roman*)}]
        \item\label{thm:permanence-i} For a model $\X$ of $X$,  the following three statements are equivalent:
            \begin{enumerate}[label=\textup{(\alph*)}]
                \item\label{thm:permanence-i-a} $\X$ is permanent,
                \item\label{thm:permanence-i-b} the special fibre $\X_s$ of $\X$ is reduced,
                \item\label{thm:permanence-i-c} all $v\in V(\X)$ are \emph{weakly unramified} over $v_K$, i.e.\ $v$ and $v_K$ have the same value group.
            \end{enumerate}
        \item\label{thm:permanence-ii} Given a model $\X$, there exists a finite extension $L|K$ such that the normalized base change of $\X$ relative to $L|K$ is permanent.
    \end{enumerate}
\end{theorem}

The proof below is modelled on~\cite[Section 2.2]{ArzdorfWewers}, adapted to the present definition of a permanent model.
Its logic is as follows.
First, a reduced special fibre remains reduced after every finite extension of the base field, and the normality criterion then shows that the corresponding base changes are normal; thus the model is permanent.
Second, reducedness of the special fibre can be checked at its generic points and is equivalent to all component valuations being weakly unramified.
Third, Epp's theorem eliminates the ramification at these finitely many generic points, so after a suitable finite extension the normalized base change has reduced special fibre and is therefore permanent.
Finally, if the original model is permanent, this normalization is already the ordinary base change; its reduced special fibre then descends to a reduced special fibre of the original model.

The proof uses Epp's theorem in the following finite-family form.
For a single discrete valuation ring, this follows from~\cite[\href{https://stacks.math.columbia.edu/tag/09II}{Proposition 09II}]{Stacks}; the original source is~\cite{Epp}. The passage to finitely many rings uses the stability of solutions under further finite extensions,
see~\cite[\href{https://stacks.math.columbia.edu/tag/0GLR}{Lemma 0GLR}]{Stacks}.
We discuss the theorem further in Section~\ref{sec:epp-and-abhyanka}.

\begin{theorem}[Epp: elimination of ramification]\label{thm:epp}
    Let $\OK\to B_i$, $i=1,\ldots,r$, be extensions of discrete valuation rings such that each $B_i$ is essentially of finite type over $\OK$, and set $F_i\coloneqq\Frac{B_i}$.
    There exists a finite extension $L|K$ with the following property.
    For every $i$ and every maximal ideal $\mathfrak n$ of the integral closure $B_{i,L}$ of $B_i$ in $(F_i\otimes_K L)_{\mathrm{red}}$, the extension of discrete valuation rings
    \[
        \OL\longrightarrow (B_{i,L})_{\mathfrak n}
    \]
    is weakly unramified, meaning that its ramification index is one, or equivalently that every uniformising parameter $\pi_L$ of $\OL$ is also a uniformising parameter of $(B_{i,L})_{\mathfrak n}$.
\end{theorem}

The following proposition proves the implication (b)$\Rightarrow$(a).

\begin{proposition} \label{prop:reducedness-implies-permanence}
    If $\X_s$ is reduced, then $\X$ is permanent.
\end{proposition}

\begin{proof}
    Let $L|K$ be a finite extension and $l|k$ the extension of the corresponding residue fields.
    The special fibre of $\X' \coloneqq \X_{\OL}$ is $\X'_s \coloneqq (\X_s)_l$.
    Since $k$ is perfect and $\X_s$ is reduced, $\X_s$ is geometrically reduced over $k$;
    see~\cite[\href{https://stacks.math.columbia.edu/tag/020I}{Lemma 020I}]{Stacks}.
    Hence $(\X_s)_l$ is reduced.
    Thus, $\X'_s$ satisfies condition~\ref{lem:conditions-for-normality-i} from Lemma~\ref{lem:conditions-for-normality}. \\
    For condition~\ref{lem:conditions-for-normality-ii}, let $\xi' \in \X'_s$ be a generic point and let $\xi\in\X_s$ be its image.
    The ring
    \[
        A'\coloneqq\OO_{\X',\xi'}
    \]
    is a localization of $\OO_{\X,\xi}\otimes_{\OK}\OL$ at the point $\xi'$.
    It is a Noetherian local domain of dimension one, and
    \begin{align}
        A'/\pi_LA'=\OO_{\X'_s,\xi'},
    \end{align}
    where $\pi_L \in \OL$ is a uniformising parameter.
    Since $\X'_s$ is reduced and $\xi'$ is a generic point, the ring on the right is a field.
    Thus, $\pi_LA'$ is the maximal ideal of $A'$.
    Moreover, $\pi_L$ is nonzero in $A'$ because $\X'$ is flat over $\OL$.
    The DVR criterion~\cite[\href{https://stacks.math.columbia.edu/tag/00PD}{Lemma 00PD}]{Stacks} now shows that $A'$ is a discrete valuation ring.
    Hence condition~\ref{lem:conditions-for-normality-ii} from Lemma~\ref{lem:conditions-for-normality} is also satisfied.
    Thus $\X'$ is normal and hence $\X$ is permanent.
\end{proof}

The following criterion relates reducedness to the component valuations; its formulation uses Proposition~\ref{prop:multiplicities-coincide}.

\begin{proposition}\label{prop:criterion-reducedness}
    Let $\X$ be a model of $X$.
    Then the following statements are equivalent.
    \begin{enumerate}[label=\textup{(\alph*)}]
        \item $\X_s$ is reduced.
        \item $\OO_{\X_s, \xi}$ is a field for every generic point $\xi \in \X_s$.
        \item Every $v_Z \in V(\X)$ is weakly unramified over $v_K$.
    \end{enumerate}
\end{proposition}

\begin{proof}
    Since $\X$ is normal, there are no embedded points in $\X_s$ by Lemma~\ref{lem:conditions-for-normality}.
    Since $\X_s$ has dimension one, it is reduced if and only if $\X_s$ is reduced in codimension zero
    which in turn is the case if $\OO_{\X_s, \xi}$ is a field for every generic point $\xi \in \X_s$. \\
    For the second equivalence, Proposition~\ref{prop:multiplicities-coincide} gives, for every generic point $\xi\in \X_s$ and $Z = \overline{\lbrace\xi\rbrace}$,
    \begin{align}
        [\Gamma_{v_Z}: \Gamma_K] = \length \OO_{\X_s, \xi}.
    \end{align}
    This implies that $\OO_{\X_s, \xi}$ is a field if and only if $v_Z$ is weakly unramified over $v_K$ for all $v_Z \in V(\X)$.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:permanence}]
    Proposition~\ref{prop:criterion-reducedness} proves the equivalence
    \ref{thm:permanence-i-b}$\Leftrightarrow$\ref{thm:permanence-i-c},
    and Proposition~\ref{prop:reducedness-implies-permanence} proves
    \ref{thm:permanence-i-b}$\Rightarrow$\ref{thm:permanence-i-a}.

    We next prove part~\ref{thm:permanence-ii}.
    Let $\xi_1,\ldots,\xi_r$ be the generic points of $\X_s$ and set
    \[
        B_i\coloneqq\OO_{\X,\xi_i}.
    \]
    These are discrete valuation rings essentially of finite type over $\OK$.
    Choose a finite extension $L|K$ as in Theorem~\ref{thm:epp}, and let
    \[
        \mathcal Y\coloneqq(\X_{\OL})^\sim
    \]
    be the normalized base change.
    The morphism $\mathcal Y\to\X$ is finite.
    Hence, if $\xi'$ is a generic point of $\mathcal Y_s$, then its image in $\X_s$ is one of the points $\xi_i$, and $\OO_{\mathcal Y,\xi'}$ is one of the localizations $(B_{i,L})_{\mathfrak n}$ occurring in Theorem~\ref{thm:epp}.
    Consequently, $\pi_L$ is a uniformising parameter of $\OO_{\mathcal Y,\xi'}$, and
    \[
        \OO_{\mathcal Y_s,\xi'}
        =\OO_{\mathcal Y,\xi'}/\pi_L\OO_{\mathcal Y,\xi'}
    \]
    is a field.
    Proposition~\ref{prop:criterion-reducedness}, applied over $L$, therefore shows that $\mathcal Y_s$ is reduced.
    The residue field of $L$ is perfect because it is a finite extension of the perfect field $k$.
    Proposition~\ref{prop:reducedness-implies-permanence}, again applied over $L$, now shows that $\mathcal Y$ is permanent.
    This proves part~\ref{thm:permanence-ii}.

    It remains to prove
    \ref{thm:permanence-i-a}$\Rightarrow$\ref{thm:permanence-i-b}.
    Suppose that $\X$ is permanent, and choose $L|K$ as in the preceding paragraph.
    Then $\X_{\OL}$ is normal, so the normalization morphism
    \[
        \mathcal Y=(\X_{\OL})^\sim\longrightarrow\X_{\OL}
    \]
    is an isomorphism.
    Hence $(\X_s)_l=\mathcal Y_s$ is reduced, where $l$ is the residue field of $L$.
    Since $l|k$ is faithfully flat, reducedness descends from $(\X_s)_l$ to $\X_s$.
    Indeed, on an affine open this follows from the injectivity of $A\to A\otimes_k l$: a nilpotent element of $A$ maps to zero because $A\otimes_k l$ is reduced, and hence was already zero in $A$.
    Thus $\X_s$ is reduced, completing the proof.
\end{proof}



\section{Extension of component valuations}\label{sec:extension-component-valuations}
To compare the special fibres before and after normalized base change, we describe how their component valuations are related by extension of valued function fields.

Let $L|K$ be a finite field extension and let $\X$ be a model of $X$.
Denote by $\X' \coloneqq (\X_{\OL})^\sim$ the normalized base change of $\X$ to $\OL$.
Let $\xi \in \X_s$ be a generic point and let $\xi' \in \X_s'$ be a generic point lying over $\xi$.
Since $\X$ and $\X'$ are normal, Lemma~\ref{lem:conditions-for-normality}\ref{lem:conditions-for-normality-ii} shows that
$R \coloneqq \OO_{\X, \xi}$ and $S \coloneqq \OO_{\X', \xi'}$ are discrete valuation rings dominating $\OK$ and $\OL$, respectively.
Thus,
\begin{enumerate}[label=\textup{(\alph*)}]
    \item $R \supset \OO_K$ (resp. $S \supset \OL$); and
    \item $\m_R \cap \OK = \pi_K\OK$ (resp. $\m_S \cap \OL = \pi_L\OL$).
\end{enumerate}

The function field after base change is the compositum
\begin{align}
    F_{X_L}=F_X\mathbin{\cdot}L.
\end{align}

\section{Eliminating ramification}\label{sec:eliminating-ramification}
The proof of Theorem~\ref{thm:permanence}\ref{thm:permanence-ii} applies Epp's theorem simultaneously to the discrete valuation rings at the generic points of $\X_s$.
After the resulting finite extension of $K$, the corresponding local rings on the normalized base change are weakly unramified over the new base ring.
Equivalently, the normalized base change has reduced special fibre.
This is the geometric form of elimination of ramification used above.


\section{Epp's theorem and Abhyanka's lemma}\label{sec:epp-and-abhyanka}
Theorem~\ref{thm:epp} states Epp's theorem in the form used above.
In the tamely ramified case, the corresponding effect of a base extension on ramification indices is expressed more explicitly by Abhyanka's lemma.

\begin{lemma}[Abhyanka]\label{lem:abhyanka}
    Let $L|K$ be a finite extension.
    Let $\lbrace v_1, \dots, v_n \rbrace = V(X)$ denote the extended valuation on the function field $F_X|K$.
    We have that $F_{X_L}=F_X\mathbin{\cdot}L$ and with the extended valuation denoted as $w_{1,1}, \dots, w_{n, j_n}$ respectively.
    Assume that $v_i|v_{{\scriptscriptstyle G}, K}$ are tamely ramified for all $i$.
    If $e_i \coloneqq e(v_i|v_{{\scriptscriptstyle G},K})$ divides
    $e(v_{{\scriptscriptstyle G}, L}, v_{{\scriptscriptstyle G}, K}) = e(v_L|v_K)$ for all $i$, then $w_{i,j} \in V(X_L)$
    is weakly unramified over $v_{{\scriptscriptstyle G}, L}$ for all $i,j$.
\end{lemma}

\section{Tame and wild examples}\label{sec:tame-and-wild-examples}
The following examples illustrate how a suitable finite extension and a change of coordinates can produce a model with reduced special fibre, and why the choice of extension matters in the wildly ramified case.

\begin{example}[Obtaining a reduced special fibre after base change]
    Let $K = \QQ_3$ and consider the projective curve
    \begin{align}
        X: \, y^3 = 3x^2 + 1
    \end{align}
    over $K$.
    The equation defines a model $\X$ over $\OK=\ZZ_3$ (cf. Example \ref{ex:model-of-plane-curve}).
    The reduction $\X_s/\FF_3$ is given by $0 = y^3 - 1 = (y - 1)^3$, so $\X_s$ is not reduced.
    The irreducible component is given by $Z: \, y-1$; denote the corresponding valuation by $v$.
    For $z \coloneqq y - 1$ we have
    \begin{align*}
        3x^2 &= y^3 - 1 \\
            &= (z+1)^3 - 1 \\
            &= z^3 + 3z^2 + 3z,
    \end{align*}
    where the right hand side is an Eisenstein polynomial.
    This gives
    \begin{align}
        v(z)=\frac{1}{3}v_{{\scriptscriptstyle G},K}(3x^2)=\frac{1}{3}\notin\ZZ=\Gamma_K.
    \end{align}
    Let us extend the ground field to $L = \QQ_3[\theta]$ where $\theta^3 - 3 = 0$ and apply the coordinate transformation
    $(x,y) \mapsto (x_1, 1+ \theta y_1)$.
    We then obtain a birational curve
    \begin{align}
        X_1: \, y_1^3 + \theta^2 y_1^2 + \theta y_1 = x_1^2.
    \end{align}
    Since $y_1^3 \equiv x_1^2$ (mod $\theta$), $\X_s$ is reduced.
    If $w$ denotes the extension of $v$ associated with this component after base change, then
    $w(\theta y_1)=w(y-1)=\frac{1}{3}=v_L(\theta)\in\Gamma_L$, and hence $w(y_1)=0$.
\end{example}

\begin{example}[Reduction after wildly ramified field extension]
    Consider the curve
    \begin{align}
        X: \, y^2 = 8x^3 - 2
    \end{align}
    defined over the field $K \coloneqq \QQ_2$.
    This example illustrates that the choice of field extension is crucial to obtaining a reduced special fibre,
    since a wildly ramified field extension is not unique.
    \begin{enumerate}[label=\textup{(\alph*)}]
        \item First let us consider the extension $L_1 = K(\theta_1)$ where $\theta_1^2 - 2 = 0$ and the coordinate transformation
            $(x,y) \mapsto (x_1, 2y_1 - \theta_1 - 2)$.
            We obtain a birational curve
            \begin{align}
                X_1: \, y_1^2 - (\theta_1 + 2)y_1 = 2x_1^3 - (2 + \theta_1).
            \end{align}
            The Newton polygon of the resulting quadratic in $y_1$ gives $v(y_1)=\frac14$ for an extension $v$ of
            $v_{{\scriptscriptstyle G},L_1}$ to $F_{X_1}$.
            Since $\Gamma_{L_1}=\frac12\ZZ$ and $[F_{X_1}:L_1(x_1)]=2$, it follows that $[\Gamma_{F_{X_1}}: \Gamma_{L_1}] = 2 \neq 1$.
        \item Let us consider the field extension $L_2 = K(\theta_2)$ where $\theta_2^2 + 2 = 0$
            and the coordinate transformation $(x,y) \mapsto (x_2, 2\theta_2 y_2 + \theta_2)$.
            We obtain a birational curve
            \begin{align}
                X_2: \, y_2^2 + y_2 = -x_2^3.
            \end{align}
            This equation is in AS form, so its special fibre is reduced.
            Thus $[\Gamma_{F_{X_2}}: \Gamma_{L_2}] = 1$.
    \end{enumerate}
\end{example}
